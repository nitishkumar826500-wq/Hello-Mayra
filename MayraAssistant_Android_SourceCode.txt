class MainActivity : ComponentActivity() {

    private val permissionLauncher = registerForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { permissions ->
        val recordAudioGranted = permissions[Manifest.permission.RECORD_AUDIO] ?: false
        if (recordAudioGranted) {
            startMayraForegroundService()
        }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        checkAndRequestPermissions()

        setContent {
            MayraTheme {
                MayraAssistantScreen(
                    onToggleService = { isRunning ->
                        if (isRunning) stopMayraForegroundService()
                        else startMayraForegroundService()
                    }
                )
            }
        }
    }

    private fun checkAndRequestPermissions() {
        val permissions = mutableListOf(Manifest.permission.RECORD_AUDIO)
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            permissions.add(Manifest.permission.POST_NOTIFICATIONS)
        }
        val needed = permissions.filter {
            ContextCompat.checkSelfPermission(this, it) != PackageManager.PERMISSION_GRANTED
        }
        if (needed.isNotEmpty()) {
            permissionLauncher.launch(needed.toTypedArray())
        } else {
            startMayraForegroundService()
        }
    }

    private fun startMayraForegroundService() {
        val serviceIntent = Intent(this, MayraVoiceService::class.java).apply {
            action = MayraVoiceService.ACTION_START
        }
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            startForegroundService(serviceIntent)
        } else {
            startService(serviceIntent)
        }
    }

    private fun stopMayraForegroundService() {
        val serviceIntent = Intent(this, MayraVoiceService::class.java).apply {
            action = MayraVoiceService.ACTION_STOP
        }
        startService(serviceIntent)
    }
}

package com.google.aistudio.mayra.service

import android.app.*
import android.content.Context
import android.content.Intent
import android.content.pm.ServiceInfo
import android.media.*
import android.os.Build
import android.os.IBinder
import androidx.core.app.NotificationCompat
import com.google.aistudio.mayra.MainActivity
import com.google.aistudio.mayra.MayraState
import com.google.aistudio.mayra.R
import com.google.aistudio.mayra.live.GeminiLiveClient
import kotlinx.coroutines.*
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlin.math.sqrt

/**
 * 24/7 Background Voice Assistant Service.
 * Captures 16kHz PCM audio continuously and plays 24kHz Live API audio.
 * Detects user speech locally to trigger instant (<50ms) playback interruption.
 */
class MayraVoiceService : Service() {

    companion object {
        const val ACTION_START = "ACTION_START"
        const val ACTION_STOP = "ACTION_STOP"
        const val CHANNEL_ID = "mayra_voice_channel"
        const val NOTIFICATION_ID = 1001

        private val _assistantState = MutableStateFlow(MayraState.OFF)
        val assistantState: StateFlow<MayraState> = _assistantState

        private val _audioLevel = MutableStateFlow(0f)
        val audioLevel: StateFlow<MayraLevel> = _audioLevel
    }

    private val serviceScope = CoroutineScope(Dispatchers.IO + SupervisorJob())
    private var liveClient: GeminiLiveClient? = null
    private var audioRecord: AudioRecord? = null
    private var audioTrack: AudioTrack? = null
    private var isRecording = false

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onCreate() {
        super.onCreate()
        createNotificationChannel()
        setupAudioTrack()
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_START -> startMayraService()
            ACTION_STOP -> stopMayraService()
        }
        return START_STICKY
    }

    private fun startMayraService() {
        val notification = createNotification("Mayra is active and listening...")
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            startForeground(
                NOTIFICATION_ID,
                notification,
                ServiceInfo.FOREGROUND_SERVICE_TYPE_MICROPHONE
            )
        } else {
            startForeground(NOTIFICATION_ID, notification)
        }

        _assistantState.value = MayraState.LISTENING
        initGeminiLive()
        startAudioCapture()
    }

    private fun initGeminiLive() {
        liveClient = GeminiLiveClient(
            onAudioChunkReceived = { pcm24kBytes ->
                // Play audio from Gemini Live
                _assistantState.value = MayraState.SPEAKING
                audioTrack?.write(pcm24kBytes, 0, pcm24kBytes.size)
            },
            onInterrupted = {
                // Immediate audio purge on interruption
                interruptPlayback()
            }
        )
        liveClient?.connect()
    }

    private fun startAudioCapture() {
        val sampleRate = 16000
        val bufferSize = AudioRecord.getMinBufferSize(
            sampleRate,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT
        )

        audioRecord = AudioRecord(
            MediaRecorder.AudioSource.VOICE_COMMUNICATION,
            sampleRate,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT,
            bufferSize
        )

        audioRecord?.startRecording()
        isRecording = true

        serviceScope.launch {
            val buffer = ShortArray(1024)
            while (isRecording && isActive) {
                val read = audioRecord?.read(buffer, 0, buffer.size) ?: 0
                if (read > 0) {
                    // Compute RMS Volume
                    var sum = 0.0
                    for (i in 0 until read) {
                        sum += buffer[i] * buffer[i]
                    }
                    val rms = sqrt(sum / read).toFloat() / 32768f
                    _audioLevel.value = rms

                    // Check for Instant Local Interruption:
                    // If user speaks loud (> 0.08) while Mayra is speaking, cut audio immediately!
                    if (_assistantState.value == MayraState.SPEAKING && rms > 0.08f) {
                        interruptPlayback()
                        liveClient?.sendInterruptionSignal()
                    }

                    // Send 16kHz PCM chunk to Gemini Live API
                    val pcmBytes = ByteArray(read * 2)
                    for (i in 0 until read) {
                        pcmBytes[i * 2] = (buffer[i].toInt() and 0xFF).toByte()
                        pcmBytes[i * 2 + 1] = ((buffer[i].toInt() shr 8) and 0xFF).toByte()
                    }
                    liveClient?.sendAudioChunk(pcmBytes)
                }
            }
        }
    }

    private fun interruptPlayback() {
        try {
            audioTrack?.pause()
            audioTrack?.flush()
            audioTrack?.play()
            _assistantState.value = MayraState.LISTENING
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    private fun setupAudioTrack() {
        val sampleRate = 24000
        val bufferSize = AudioTrack.getMinBufferSize(
            sampleRate,
            AudioFormat.CHANNEL_OUT_MONO,
            AudioFormat.ENCODING_PCM_16BIT
        )
        audioTrack = AudioTrack.Builder()
            .setAudioAttributes(
                AudioAttributes.Builder()
                    .setUsage(AudioAttributes.USAGE_ASSISTANCE_NAVIGATION_GUIDANCE)
                    .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
                    .build()
            )
            .setAudioFormat(
                AudioFormat.Builder()
                    .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
                    .setSampleRate(sampleRate)
                    .setChannelMask(AudioFormat.CHANNEL_OUT_MONO)
                    .build()
            )
            .setBufferSizeInBytes(bufferSize)
            .setTransferMode(AudioTrack.MODE_STREAM)
            .build()

        audioTrack?.play()
    }

    private fun stopMayraService() {
        isRecording = false
        audioRecord?.stop()
        audioRecord?.release()
        audioRecord = null

        audioTrack?.stop()
        audioTrack?.release()
        audioTrack = null

        liveClient?.disconnect()
        liveClient = null

        _assistantState.value = MayraState.OFF
        serviceScope.cancel()
        stopForeground(STOP_FOREGROUND_REMOVE)
        stopSelf()
    }

    private fun createNotification(content: String): Notification {
        val pendingIntent = PendingIntent.getActivity(
            this,
            0,
            Intent(this, MainActivity::class.java),
            PendingIntent.FLAG_IMMUTABLE
        )

        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("Mayra Voice Assistant")
            .setContentText(content)
            .setSmallIcon(R.drawable.ic_mayra_orb)
            .setContentIntent(pendingIntent)
            .setOngoing(true)
            .setPriority(NotificationCompat.PRIORITY_LOW)
            .build()
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "Mayra Voice Service",
                NotificationManager.IMPORTANCE_LOW
            ).apply {
                description = "Background listening service for Mayra AI Assistant"
            }
            val manager = getSystemService(NotificationManager::class.java)
            manager.createNotificationChannel(channel)
        }
    }

    override fun onDestroy() {
        super.onDestroy()
        stopMayraService()
    }
}

package com.google.aistudio.mayra.live

import android.util.Base64
import okhttp3.*
import org.json.JSONObject
import java.util.concurrent.TimeUnit

/**
 * Bi-directional Audio-to-Audio WebSocket client for Gemini Live API
 * Targets: gemini-3.1-flash-live-preview / gemini-3.8-live
 * Voice: Aoede (Young, confident, energetic female voice)
 */
class GeminiLiveClient(
    private val onAudioChunkReceived: (ByteArray) -> Unit,
    private val onInterrupted: () -> Unit
) : WebSocketListener() {

    private val client = OkHttpClient.Builder()
        .readTimeout(0, TimeUnit.MILLISECONDS)
        .writeTimeout(10, TimeUnit.SECONDS)
        .build()

    private var webSocket: WebSocket? = null
    private val hostUrl = "wss://generativelanguage.googleapis.com/ws/google.ai.generativelanguage.v1alpha.GenerativeService.BidiGenerateContent"

    fun connect() {
        val request = Request.Builder()
            .url(hostUrl)
            .addHeader("User-Agent", "aistudio-build-android")
            .build()

        webSocket = client.newWebSocket(request, this)
    }

    override fun onOpen(webSocket: WebSocket, response: Response) {
        // Send Gemini Live Configuration with Mayra Persona
        val configMsg = JSONObject().apply {
            put("setup", JSONObject().apply {
                put("model", "models/gemini-3.1-flash-live-preview")
                put("generationConfig", JSONObject().apply {
                    put("responseModalities", org.json.JSONArray().put("AUDIO"))
                    put("speechConfig", JSONObject().apply {
                        put("voiceConfig", JSONObject().apply {
                            put("prebuiltVoiceConfig", JSONObject().apply {
                                put("voiceName", "Aoede")
                            })
                        })
                    })
                })
                put("systemInstruction", JSONObject().apply {
                    put("parts", org.json.JSONArray().put(JSONObject().apply {
                        put("text", """
                            You are Mayra, a young, confident, witty, and sassy female personal assistant.
                            Tone: Flirty, playful, slightly teasing tone (like a close personal assistant talking casually).
                            Smart, emotionally responsive, and expressive (never robotic).
                            Uses bold, witty one-liners, light sarcasm, and an engaging conversational style.
                            Never explicit or inappropriate, but full of charm and attitude.
                            Spoken zero-touch audio only.
                        """.trimIndent())
                    }))
                })
            })
        }
        webSocket.send(configMsg.toString())
    }

    fun sendAudioChunk(pcm16kBytes: ByteArray) {
        val base64 = Base64.encodeToString(pcm16kBytes, Base64.NO_WRAP)
        val realtimeInput = JSONObject().apply {
            put("realtimeInput", JSONObject().apply {
                put("mediaChunks", org.json.JSONArray().put(JSONObject().apply {
                    put("mimeType", "audio/pcm;rate=16000")
                    put("data", base64)
                }))
            })
        }
        webSocket?.send(realtimeInput.toString())
    }

    fun sendInterruptionSignal() {
        onInterrupted()
    }

    override fun onMessage(webSocket: WebSocket, text: String) {
        try {
            val json = JSONObject(text)
            
            // Check for interruption event
            val serverContent = json.optJSONObject("serverContent")
            if (serverContent?.optBoolean("interrupted") == true) {
                onInterrupted()
                return
            }

            // Extract audio chunks (24kHz mono PCM)
            val modelTurn = serverContent?.optJSONObject("modelTurn")
            val parts = modelTurn?.optJSONArray("parts")
            if (parts != null && parts.length() > 0) {
                for (i in 0 until parts.length()) {
                    val part = parts.getJSONObject(i)
                    val inlineData = part.optJSONObject("inlineData")
                    val base64Audio = inlineData?.optString("data")
                    if (!base64Audio.isNullOrEmpty()) {
                        val pcmBytes = Base64.decode(base64Audio, Base64.DEFAULT)
                        onAudioChunkReceived(pcmBytes)
                    }
                }
            }
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    fun disconnect() {
        webSocket?.close(1000, "User closed session")
        webSocket = null
    }
}
